\paragraph{Tiền xử lý và Kỹ thuật đặc trưng} \hfill \\
Để máy tính có thể hiểu được các đặc trưng vật lý, chúng ta cần chuyển đổi các chuỗi văn bản thành giá trị số.

\begin{lstlisting}[language=Python, caption=Tach so va trich xuat dac trung]
# Tach so tu chuoi (vi du: '23.4 kmpl' -> 23.4)
def extract_num(text):
    if pd.isna(text): return np.nan
    res = re.findall(r'(\d+\.?\d*)', str(text))
    return float(res[0]) if res else np.nan

df['mileage'] = df['mileage'].apply(extract_num)
df['engine'] = df['engine'].apply(extract_num)
df['max_power'] = df['max_power'].apply(extract_num)

# Trich xuat Brand tu ten xe
df['brand'] = df['name'].str.split().str[0]
\end{lstlisting}
